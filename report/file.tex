\section{VINS-Mono}

\begin{frame}{VINS-Mono: Overview \& Preprocessing}
  \head{Optimization-based, 4 subsystems}
  \begin{itemize}
    \item Measurement preprocessing
    \item Loosely-coupled estimator initialization
    \item Tightly-coupled sliding-window VIO
    \item Relocalization \& global pose graph optimization
  \end{itemize}
  \head{Visual front-end}
  \begin{itemize}
    \item KLT sparse optical flow for feature tracking
    \item Min.\ pixel separation $\rightarrow$ uniform feature distribution
    \item Keyframe if: avg.\ parallax above threshold, or too few tracked features
  \end{itemize}
  \head{IMU pre-integration}
  \begin{itemize}
    \item Many high-rate IMU readings $\rightarrow$ one relative-motion block between frames
  \end{itemize}
\end{frame}

\begin{frame}{VINS-Mono: Estimator Initialization}
  \head{Goal}
  \begin{itemize}
    \item Up-to-scale, vision-only structure $\rightarrow$ metric state vector
    \item Sequential solve: gyro bias $\rightarrow$ scale $\rightarrow$ gravity $\rightarrow$ velocity
  \end{itemize}
  \head{Vision-only SfM (sliding window)}
  \begin{itemize}
    \item Five-point algorithm $\rightarrow$ PnP $\rightarrow$ global BA
    \item Output: up-to-scale camera poses + triangulated 3D features
  \end{itemize}
  \head{Visual-inertial alignment}
  \begin{itemize}
    \item Up-to-scale poses aligned with IMU pre-integration
    \item Output: metric state variables
  \end{itemize}
\end{frame}

\begin{frame}{VINS-Mono: Tightly-Coupled Sliding-Window VIO}
  \head{MAP nonlinear optimization, jointly over}
  \begin{itemize}
    \item Recent keyframe states
    \item Camera--IMU extrinsics
    \item 3D feature positions
  \end{itemize}
  \head{Cost function}
  \begin{itemize}
    \item IMU residual: predicted motion between frames vs.\ pre-integrated IMU
    \item Visual residual: observed vs.\ predicted feature location
  \end{itemize}
  \head{Marginalization}
  \begin{itemize}
    \item Schur complement $\rightarrow$ old information kept as prior for future windows
  \end{itemize}
\end{frame}

\begin{frame}{VINS-Mono: Relocalization \& Pose Graph}
  \head{Relocalization}
  \begin{itemize}
    \item Current window aligned to previously visited places
    \item Drift only in 4-DOF ($x, y, z$, yaw); roll \& pitch observable via gravity
    \item Loop detection: DBoW2 + BRIEF descriptors
    \item VIO cost + extra loop-closure feature term
  \end{itemize}
  \head{Global pose graph optimization}
  \begin{itemize}
    \item Asynchronous, whole past trajectory
    \item Global consistency
    \item Spatial downsampling of redundant keyframes $\rightarrow$ bounded compute
  \end{itemize}
\end{frame}

\section{DPVO}

\begin{frame}{DPVO: Overview \& Representation}
  \head{Learning-based monocular VO, 3 parts}
  \begin{itemize}
    \item Feature \& patch extraction
    \item Recurrent update operator
    \item Differentiable bundle adjustment
  \end{itemize}
  \head{Scene representation}
  \begin{itemize}
    \item Camera poses $T \in SE(3)^N$ + sparse square image patches
    \item Patch: pixel coordinates + inverse depth $d$, constant depth per patch
      \[ P_k = [x, y, 1, d]^T \in \mathbb{R}^{4 \times p^2} \]
    \item Patch graph: edges between patches and nearby frames
    \item Projection of patch $k$ (from frame $i$) into frame $j$:
      \[ P_{kj}' \sim K \, T_j \, T_i^{-1} K^{-1} P_k \]
  \end{itemize}
\end{frame}

\begin{frame}{DPVO: Feature Extraction \& Update Operator}
  \head{Front-end}
  \begin{itemize}
    \item Two ResNets: context features + matching features
    \item Random pixel sampling $\rightarrow$ $p \times p$ patch crops 
  \end{itemize}
  \head{Recurrent update operator}
  \begin{itemize}
    \item Correlation: patch vs.\ target-frame features, $7 \times 7$ window around prediction
      \[ C_{uv\alpha\beta} = \langle g_{uv}, \, f(P_{kj}'(u,v) + \Delta_{\alpha\beta}) \rangle \]
    \item Temporal convolution + Spatial aggregation
    \item Transition block: gated memory, stable over iterations
    \item Factor head: flow correction $\delta_{kj}$ + confidence $\Sigma_{kj}$
  \end{itemize}
\end{frame}

\begin{frame}{DPVO: Differentiable BA \& System Logic}
  \head{Differentiable bundle adjustment}
  \begin{itemize}
    \item Flow corrections $\rightarrow$ consistent poses \& depths
      \[ \sum_{(k,j)} \big\| \omega_{ij}(T, P_k) - (\hat{P}_{kj}' + \delta_{kj}) \big\|_{\Sigma_{kj}}^2 \]
    \item Gauss--Newton + Schur complement
    \item Fully differentiable $\rightarrow$ end-to-end training
  \end{itemize}

\section{OpenVINS}
 

\begin{frame}{OpenVINS: Method Overview}
  \head{MSCKF (EKF-based VIO)}
  \begin{itemize}
    \item State: IMU + sliding window of past poses
  \end{itemize}

  \head{Propagation}
  \begin{itemize}
    \item IMU integration $\rightarrow$ predicted pose
  \end{itemize}

  \head{EKF update}
  \begin{itemize}
    \item Feature tracking 
    \item Track ends $\rightarrow$ triangulate point
    \item Reprojection vs.\ measurement $\rightarrow$ residual
    \item Correct poses + IMU state
    \item Features not in state $\rightarrow$ small, fast filter
  \end{itemize}
\end{frame}
